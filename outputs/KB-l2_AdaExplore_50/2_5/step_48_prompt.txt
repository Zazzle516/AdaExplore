import torch
import torch.nn as nn
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _conv_transpose2d_kernel(
    x_ptr, w_ptr, b_ptr, out_ptr,
    N, IC, IH, IW,
    OC, OH, OW,
    KH: tl.constexpr, KW: tl.constexpr,
    STRIDE: tl.constexpr, PAD: tl.constexpr,
    BLOCK_OC: tl.constexpr, BLOCK_SP: tl.constexpr,
):
    pid_sp = tl.program_id(0)
    pid_oc = tl.program_id(1)
    pid_n = tl.program_id(2)

    sp_offs = pid_sp * BLOCK_SP + tl.arange(0, BLOCK_SP)
    oc_offs = pid_oc * BLOCK_OC + tl.arange(0, BLOCK_OC)

    sp_mask = sp_offs < (OH * OW)
    oc_mask = oc_offs < OC

    oh = sp_offs // OW
    ow = sp_offs % OW

    acc = tl.zeros((BLOCK_SP, BLOCK_OC), dtype=tl.float32)

    # For ConvTranspose2d: out[n, oc, oh, ow] = sum_{ic, kh, kw} x[n, ic, ih, iw] * w[ic, oc, kh, kw]
    # where ih*stride - pad + kh = oh, so ih = (oh + pad - kh) / stride
    # Pre-transposed weight: w_t[oc, ic, kh, kw]
    for kh in tl.static_range(0, KH):
        for kw in tl.static_range(0, KW):
            ih_num = oh + PAD - kh
            iw_num = ow + PAD - kw
            ih = ih_num // STRIDE
            iw = iw_num // STRIDE
            valid = (ih_num % STRIDE == 0) & (iw_num % STRIDE == 0) & \
                    (ih >= 0) & (ih < IH) & (iw >= 0) & (iw < IW) & sp_mask

            for ic in range(0, IC):
                x_off = pid_n * IC * IH * IW + ic * IH * IW + ih * IW + iw
                x_val = tl.load(x_ptr + x_off, mask=valid, other=0.0)  # [BLOCK_SP]

                w_off = oc_offs * IC * KH * KW + ic * KH * KW + kh * KW + kw
                w_val = tl.load(w_ptr + w_off, mask=oc_mask, other=0.0)  # [BLOCK_OC]

                acc += x_val[:, None] * w_val[None, :]

    # bias from conv_transpose + subtract self.bias + tanh
    b_val = tl.load(b_ptr + oc_offs, mask=oc_mask, other=0.0)
    acc = acc + b_val[None, :]
    e2x = tl.exp(2.0 * acc)
    out = (e2x - 1.0) / (e2x + 1.0)

    # Store: out[n, oc, oh, ow], layout NCHW
    out_off = pid_n * OC * OH * OW + oc_offs[None, :] * OH * OW + sp_offs[:, None]
    mask = sp_mask[:, None] & oc_mask[None, :]
    tl.store(out_ptr + out_off, out, mask=mask)

- The current gather-based ConvTranspose2d does redundant work: for stride=2, only 1/4 of (kh,kw) combinations are valid per output pixel, but the inner static loop still issues all 16 (kh,kw) loads. Split the kernel into 4 sub-kernels (or use stride-aware indexing) keyed on (oh%2, ow%2) so each output only loops over the (KH/STRIDE)*(KW/STRIDE)=2*2=4 valid taps instead of 16 — a ~4x reduction in loads and dot work.
- Tile the spatial dimension on contiguous OW rows (e.g. BLOCK_OW=64, BLOCK_OH=1 or 2) instead of flattening OH*OW; this makes `iw` contiguous and lets x-loads coalesce across BLOCK_SP for fixed ih. Add autotune configs with BLOCK_OC=128, BLOCK_OW=128, num_warps=8, num_stages=3.
- Cache the permuted weight (`_weight_t`) once in __init__ as a registered buffer rather than recomputing `.permute().contiguous()` every forward call.
