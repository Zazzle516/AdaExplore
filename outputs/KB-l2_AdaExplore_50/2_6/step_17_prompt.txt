def fused_conv_softmax_pool_kernel(
    x_ptr, w_ptr, b_ptr, out_ptr,
    N, IC: tl.constexpr,
    D_in, H_in, W_in,
    D_conv, H_conv, W_conv,
    OD, OH, OW,
    OC: tl.constexpr,
    KD: tl.constexpr, KH: tl.constexpr, KW: tl.constexpr,
    P: tl.constexpr,
):
    # one program per (n, od, oh, ow); computes OC outputs
    pid = tl.program_id(0)
    ow = pid % OW
    tmp = pid // OW
    oh = tmp % OH
    tmp = tmp // OH
    od = tmp % OD
    n = tmp // OD

    # pre-pool position start
    d_start = od * P
    h_start = oh * P
    w_start = ow * P

    oc_offs = tl.arange(0, OC)

    # accumulator for max over P*P*P window per channel
    max_vals = tl.full([OC], -float('inf'), dtype=tl.float32)

    # bias load (constant per kernel)
    bias = tl.load(b_ptr + oc_offs)  # [OC]

    # iterate over pooling window (P^3 conv outputs)
    for di in tl.static_range(P):
        for hi in tl.static_range(P):
            for wi in tl.static_range(P):
                d_conv = d_start + di
                h_conv = h_start + hi
                w_conv = w_start + wi

                # Compute conv for all OC at this (n, d_conv, h_conv, w_conv)
                acc = bias  # [OC]

                # input region: d_conv .. d_conv+KD-1
                for kd in tl.static_range(KD):
                    d_in = d_conv + kd
                    for kh in tl.static_range(KH):
                        h_in = h_conv + kh
                        for kw in tl.static_range(KW):
                            w_in = w_conv + kw
                            # load IC values from input
                            ic_offs = tl.arange(0, IC)
                            x_off = (n * IC * D_in * H_in * W_in
                                     + ic_offs * (D_in * H_in * W_in)
                                     + d_in * H_in * W_in
                                     + h_in * W_in
                                     + w_in)
                            x_vals = tl.load(x_ptr + x_off)  # [IC]

                            # weight layout: (OC, IC, KD, KH, KW)

                + oh * OW
                + ow)

    h_start = oh * P
