    e = tl.exp(v - m[None, :])
    e = tl.where(mask_oc[:, None], e, 0.0)
    s = tl.sum(e, axis=0)
    sm = e / s[None, :]

    out = t * scaling_factor

               + offs_oc[:, None] * H * W
               + offs_hw[None, :])
    mask_store = mask_oc[:, None] & mask_hw[None, :]
    tl.store(out_ptr + out_off, out, mask=mask_store)
